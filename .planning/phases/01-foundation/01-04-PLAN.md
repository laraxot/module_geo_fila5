    Pattern:
    ```php
    class GenerateAuditReportAction extends Action
    {
        public function execute(AuditReportRequest $request): AuditReport
        {
            $query = Activity::query()
                ->inDateRange($request->date_from, $request->date_to)
                ->forSubject($request->subject_type)
                ->byCauser($request->user_id);
            
            return new AuditReport([
                'summary' => $this->buildSummary($query),
                'timeline' => $this->buildTimeline($query->get()),
                'by_user' => $this->breakdownByUser($query->get()),
            ]);
        }
    }
    ```
  </action>
